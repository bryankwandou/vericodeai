    function getCaretCoordinates(element, position) {
      const container = element.parentElement;
      const mirrorId = 'input-mirror-' + element.id;
      let mirrorDiv = document.getElementById(mirrorId);
      if (!mirrorDiv) {
        mirrorDiv = document.createElement('div');
        mirrorDiv.id = mirrorId;
        mirrorDiv.className = 'input-mirror-div';
        mirrorDiv.style.position = 'absolute';
        mirrorDiv.style.visibility = 'hidden';
        mirrorDiv.style.pointerEvents = 'none';
        container.appendChild(mirrorDiv);
      }

      const style = mirrorDiv.style;
      const computed = window.getComputedStyle(element);

      const properties = [
        'fontFamily', 'fontSize', 'fontWeight', 'fontStyle', 'lineHeight',
        'paddingLeft', 'paddingRight', 'paddingTop', 'paddingBottom',
        'borderLeftWidth', 'borderRightWidth', 'borderTopWidth', 'borderBottomWidth',
        'borderStyle', 'boxSizing', 'textAlign', 'textTransform', 'letterSpacing', 'wordSpacing'
      ];
      properties.forEach(prop => {
        style[prop] = computed[prop];
      });

      style.whiteSpace = 'pre-wrap';
      style.wordWrap = 'break-word';
      style.width = element.clientWidth + 'px';
      style.height = element.clientHeight + 'px';

      // Position the mirror directly over the element
      style.left = element.offsetLeft + 'px';
      style.top = element.offsetTop + 'px';

      const value = element.value;
      mirrorDiv.textContent = value.substring(0, position);

      const span = document.createElement('span');
      span.textContent = '|';
      mirrorDiv.appendChild(span);

      return {
        top: element.offsetTop + span.offsetTop - element.scrollTop,
        left: element.offsetLeft + span.offsetLeft - element.scrollLeft,
        height: span.offsetHeight
      };
    }

    let caretBlinkTimeout = null;
    let rafId = null;
    function updateCaret() {
      const caret = document.getElementById('ai-custom-caret');
      if (!aiInput || !caret) return;

      if (rafId) cancelAnimationFrame(rafId);
      rafId = requestAnimationFrame(() => {
        if (document.activeElement !== aiInput) {
          caret.style.opacity = '0';
          return;
        }

        const pos = aiInput.selectionStart;
        const coordinates = getCaretCoordinates(aiInput, pos);

        caret.style.left = coordinates.left + 'px';
        caret.style.top = (coordinates.top + (coordinates.height - 18) / 2) + 'px';
        caret.style.height = '18px';
        caret.style.opacity = '1';

        caret.classList.remove('blink');
        clearTimeout(caretBlinkTimeout);
        caretBlinkTimeout = setTimeout(() => {
          if (document.activeElement === aiInput) {
            caret.classList.add('blink');
          }
        }, 500);
      });
    }

    function generateSVG(variant, start) {
      if (variant === "circle-blur") {
        if (start === "center") {
          return `data:image/svg+xml,<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 40 40"><defs><filter id="blur"><feGaussianBlur stdDeviation="2"/></filter></defs><circle cx="20" cy="20" r="18" fill="white" filter="url(%23blur)"/></svg>`;
        }
        const { cx, cy } = getPositionCoords(start);
        return `data:image/svg+xml,<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 40 40"><defs><filter id="blur"><feGaussianBlur stdDeviation="2"/></filter></defs><circle cx="${cx}" cy="${cy}" r="18" fill="white" filter="url(%23blur)"/></svg>`;
      }

      if (start === "center" || variant === "rectangle") return "";

      const { cx, cy } = getPositionCoords(start);
      if (variant === "circle") {
        return `data:image/svg+xml,<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 40 40"><circle cx="${cx}" cy="${cy}" r="20" fill="white"/></svg>`;
      }
      return "";
    }

    function applyRandomTransition() {
      const variants = ['circle', 'circle-blur', 'rectangle', 'polygon'];
      const variant = variants[Math.floor(Math.random() * variants.length)];
      
      let start = 'center';
      if (variant === 'circle' || variant === 'circle-blur') {
        const positions = ['center', 'top-left', 'top-right', 'bottom-left', 'bottom-right', 'top-center', 'bottom-center'];
        start = positions[Math.floor(Math.random() * positions.length)];
      } else if (variant === 'rectangle') {
        const positions = ['bottom-up', 'top-down', 'left-right', 'right-left', 'top-left', 'top-right', 'bottom-left', 'bottom-right'];
        start = positions[Math.floor(Math.random() * positions.length)];
      } else if (variant === 'polygon') {
        const positions = ['top-left', 'top-right'];
        start = positions[Math.floor(Math.random() * positions.length)];
      }
      
      const blur = Math.random() > 0.5;
      
      const anim = createAnimation(variant, start, blur);
      
      let styleEl = document.getElementById(styleId);
      if (!styleEl) {
        styleEl = document.createElement('style');
        styleEl.id = styleId;
        document.head.appendChild(styleEl);
      }
      styleEl.textContent = anim.css;
